cmake_minimum_required(VERSION 3.24)
project(OpenXRFovExtension VERSION 0.1.5 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FOV_BUILD_GUI "Build the Qt6 manager" ON)
option(FOV_BUILD_TESTS "Build tests" ON)
set(OPENXR_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/OpenXR-SDK/include" CACHE PATH "OpenXR SDK include directory")
if(NOT EXISTS "${OPENXR_INCLUDE_DIR}/openxr/openxr_loader_negotiation.h")
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 or set OPENXR_INCLUDE_DIR")
endif()
set(VULKAN_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/Vulkan-Headers/include" CACHE PATH "Vulkan headers")
if(NOT EXISTS "${VULKAN_INCLUDE_DIR}/vulkan/vulkan.h")
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 or set VULKAN_INCLUDE_DIR")
endif()
find_program(FOV_GLSLANG NAMES glslang glslangValidator HINTS "${CMAKE_CURRENT_SOURCE_DIR}/external/glslang/bin" "$ENV{VULKAN_SDK}/Bin" REQUIRED)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE XR_NO_PROTOTYPES XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 XR_USE_GRAPHICS_API_D3D12 XR_USE_GRAPHICS_API_VULKAN VK_NO_PROTOTYPES)
add_compile_options(/W4 /EHsc /utf-8)
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${OPENXR_INCLUDE_DIR} ${VULKAN_INCLUDE_DIR})
add_library(fov_common STATIC common/ConfigManager.cpp common/RegistryManager.cpp)
target_link_libraries(fov_common PUBLIC advapi32 shell32)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/VulkanShader.h"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/generated"
  COMMAND "${FOV_GLSLANG}" --target-env vulkan1.1 --spirv-val -V "${CMAKE_CURRENT_SOURCE_DIR}/shaders/Vulkan.comp" -o "${CMAKE_CURRENT_BINARY_DIR}/generated/Vulkan.spv"
  COMMAND ${CMAKE_COMMAND} "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/generated/Vulkan.spv" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/generated/VulkanShader.h" -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed-spirv.cmake"
  DEPENDS shaders/Vulkan.comp scripts/embed-spirv.cmake VERBATIM)
add_library(fov_renderers STATIC renderer/D3D11EdgeRenderer.cpp renderer/D3D12EdgeRenderer.cpp renderer/D3D12FFR.cpp renderer/Shader.cpp
  renderer/VulkanContext.cpp renderer/VulkanEdgeRenderer.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/VulkanShader.h")
target_link_libraries(fov_renderers PUBLIC d3d11 d3d12 dxgi d3dcompiler)
file(READ shaders/EdgeFill.hlsl FOV_SHADER)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/EdgeFill.hlsl shaders/Temporal.hlsl)
file(READ shaders/Temporal.hlsl TEMPORAL_SHADER)
string(REPLACE "// TEMPORAL_IMPL" "${TEMPORAL_SHADER}" FOV_SHADER "${FOV_SHADER}")
configure_file(renderer/ShaderSource.h.in generated/ShaderSource.h @ONLY)
target_include_directories(fov_renderers PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_library(OpenXRFovExtension SHARED layer/OpenXRLayer.cpp layer/IPCServer.cpp)
target_link_libraries(OpenXRFovExtension PRIVATE fov_common fov_renderers)
set_target_properties(OpenXRFovExtension PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(FovLayerControl common/LayerControl.cpp)
target_link_libraries(FovLayerControl PRIVATE fov_common)
set_target_properties(FovLayerControl PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(FOV_BUILD_GUI)
  enable_language(RC)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
  configure_file(gui/Manager.rc.in generated/Manager.rc @ONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/generated/Manager.rc" PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/gui/assets/app.ico")
  add_executable(OpenXRFovManager WIN32 gui/main.cpp gui/MainWindow.cpp gui/FFRPanel.cpp gui/Localization.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/generated/Manager.rc")
  qt_add_resources(OpenXRFovManager translations PREFIX "/i18n" FILES gui/translations.json)
  qt_add_resources(OpenXRFovManager application_icon PREFIX "/icons" FILES "Neon VR Focus Grid.png")
  target_link_libraries(OpenXRFovManager PRIVATE fov_common Qt6::Widgets Qt6::Network)
  target_compile_definitions(OpenXRFovManager PRIVATE FOV_VERSION="${PROJECT_VERSION}")
  set_target_properties(OpenXRFovManager PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
foreach(target OpenXRFovExtension FovLayerControl)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_SOURCE_DIR}/manifest/OpenXRFovExtension.json" "${CMAKE_SOURCE_DIR}/install_layer.bat" "${CMAKE_SOURCE_DIR}/uninstall_layer.bat" "$<TARGET_FILE_DIR:${target}>")
endforeach()
if(FOV_BUILD_TESTS)
  enable_testing()
  add_executable(ffr_tests tests/FFRTests.cpp)
  target_link_libraries(ffr_tests PRIVATE fov_common)
  add_test(NAME ffr_projection_and_config COMMAND ffr_tests)
  add_executable(ffr_hardware_tests tests/FFRHardwareTests.cpp)
  target_link_libraries(ffr_hardware_tests PRIVATE fov_common fov_renderers)
  add_test(NAME ffr_d3d12_hardware COMMAND ffr_hardware_tests "$<TARGET_FILE:OpenXRFovExtension>")
  set_tests_properties(ffr_d3d12_hardware PROPERTIES TIMEOUT 180 SKIP_RETURN_CODE 77)
  add_executable(fov_tests tests/CoreTests.cpp)
  target_link_libraries(fov_tests PRIVATE fov_common fov_renderers)
  add_test(NAME core_and_gpu COMMAND fov_tests)
  add_executable(layer_tests tests/LayerTests.cpp)
  target_link_libraries(layer_tests PRIVATE fov_common d3d11 d3d12 dxgi)
  add_test(NAME layer_dispatch COMMAND layer_tests "$<TARGET_FILE:OpenXRFovExtension>")
  add_test(NAME layer_dispatch_d3d12 COMMAND layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --d3d12)
  add_test(NAME layer_temporal_d3d11 COMMAND layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --temporal)
  add_test(NAME layer_temporal_d3d12 COMMAND layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --d3d12 --temporal)
  set_tests_properties(layer_temporal_d3d11 layer_temporal_d3d12 PROPERTIES TIMEOUT 30)
  set_tests_properties(layer_dispatch layer_dispatch_d3d12 PROPERTIES TIMEOUT 20)
  add_executable(LoaderProbe tests/LoaderProbe.cpp)
  target_link_libraries(LoaderProbe PRIVATE fov_common)
  add_executable(temporal_tests tests/TemporalTests.cpp)
  target_link_libraries(temporal_tests PRIVATE fov_common fov_renderers)
  add_test(NAME temporal_d3d11 COMMAND temporal_tests)
  add_test(NAME temporal_d3d12 COMMAND temporal_tests --d3d12)
  set_tests_properties(temporal_d3d11 temporal_d3d12 PROPERTIES TIMEOUT 90)
  add_executable(vulkan_tests tests/VulkanTests.cpp)
  target_link_libraries(vulkan_tests PRIVATE fov_common fov_renderers)
  add_test(NAME vulkan_gpu COMMAND vulkan_tests)
  set_tests_properties(vulkan_gpu PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
  add_executable(vulkan_layer_tests tests/VulkanLayerTests.cpp)
  target_link_libraries(vulkan_layer_tests PRIVATE fov_common fov_renderers)
  add_test(NAME vulkan_layer_array COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>")
  add_test(NAME vulkan_layer_enable2 COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --vulkan2)
  add_test(NAME vulkan_layer_separate COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --separate)
  add_test(NAME vulkan_layer_packed COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --packed)
  add_test(NAME vulkan_layer_depth COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --temporal)
  add_test(NAME vulkan_layer_depth_packed COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --temporal --packed)
  add_test(NAME vulkan_layer_depth_separate COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --temporal --separate)
  add_test(NAME vulkan_layer_mixed_extensions COMMAND vulkan_layer_tests "$<TARGET_FILE:OpenXRFovExtension>" --mixed)
  set_tests_properties(vulkan_layer_array vulkan_layer_enable2 vulkan_layer_separate vulkan_layer_packed vulkan_layer_depth vulkan_layer_depth_packed vulkan_layer_depth_separate vulkan_layer_mixed_extensions PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
  if(FOV_BUILD_GUI)
    add_executable(gui_tests tests/GuiTests.cpp gui/MainWindow.cpp gui/FFRPanel.cpp gui/Localization.cpp)
    qt_add_resources(gui_tests test_translations PREFIX "/i18n" FILES gui/translations.json)
    target_link_libraries(gui_tests PRIVATE fov_common Qt6::Widgets Qt6::Network)
    target_compile_definitions(gui_tests PRIVATE FOV_VERSION="${PROJECT_VERSION}")
    add_test(NAME gui_localization COMMAND gui_tests)
    set_tests_properties(gui_localization PROPERTIES
      ENVIRONMENT "QT_QPA_PLATFORM=minimal;QT_PLUGIN_PATH=$<TARGET_FILE_DIR:Qt6::Core>/../plugins"
      ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>" TIMEOUT 30)
  endif()
endif()
